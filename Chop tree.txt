	game:HttpGetAsync("https://raw.githubusercontent.com/3345-c-a-t-s-u-s/NOTHING/main/source.lua")
)()

-- ===== WINDOW =====D
local Windows = NothingLibrary.new({
	Title = "AKIHUB",
	Description = "🌴 Chop Your Tree",
	Keybind = Enum.KeyCode.LeftControl,
	Logo = "http://www.roblox.com/asset/?id=106070727897225"
})

-- ===== SERVICES =====
local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local RunService = game:GetService("RunService")
local TeleportService = game:GetService("TeleportService")

local LocalPlayer = Players.LocalPlayer

-- ===== SAFE WAIT MAP =====
local Plot = workspace:WaitForChild("Plots"):WaitForChild("Plot")
local HitBox = Plot
	:WaitForChild("PlotContents")
	:WaitForChild("Tree")
	:WaitForChild("HitBox")

-- ===== REMOTES (แยกชื่อ ไม่ชน) =====
local AxeRemote = ReplicatedStorage
	:WaitForChild("Remotes")
	:WaitForChild("AxeSwing")

local TapRemote = ReplicatedStorage
	:WaitForChild("Remotes")
	:WaitForChild("TapButtonClick")

-- ===== GLOBAL CONTROL =====
_G.CutTree = false
_G.CutTreeSession = 0
_G.AutoTap = false

-- ======================================================
-- ======================= TAB 1 ========================
-- ======================================================
local FarmTab = Windows:NewTab({
	Title = "Farm",
	Description = "Farm Tab",
	Icon = "rbxassetid://109805612001451"
})

local FarmSection = FarmTab:NewSection({
	Title = "Farm Menu",
	Icon = "rbxassetid://7743869054",
	Position = "Left"
})

local InfoSection = FarmTab:NewSection({
	Title = "Information",
	Icon = "rbxassetid://7733964719",
	Position = "Right"
})

-- ===== CUT TREE =====
FarmSection:NewToggle({
	Title = "Cut Tree",
	Default = false,
	Callback = function(state)
		_G.CutTree = state
		_G.CutTreeSession += 1
		local session = _G.CutTreeSession

		if not state then
			warn("Cut Tree : OFF")
			return
		end

		warn("Cut Tree : ON")

		task.spawn(function()
			local FixedCFrame = HitBox.CFrame * CFrame.new(-1, -5, 4)

			while _G.CutTree and _G.CutTreeSession == session do
				pcall(function()
					local char = LocalPlayer.Character
					if not char then return end

					local hrp = char:FindFirstChild("HumanoidRootPart")
					if not hrp then return end

					hrp.CFrame = FixedCFrame

					local axe =
						char:FindFirstChild("Axe")
						or LocalPlayer.Backpack:FindFirstChild("Axe")

					if axe then
						axe.Parent = char
						axe:Activate()
					end

					AxeRemote:FireServer()
				end)

				task.wait(0.001)
			end
		end)
	end
})

-- ===== AUTO WATER PUMP =====
FarmSection:NewToggle({
	Title = "Auto Water Pump",
	Default = false,
	Callback = function(state)
		_G.AutoTap = state

		if not state then return end

		task.spawn(function()
			while _G.AutoTap do
				pcall(function()
					TapRemote:FireServer(Plot)
				end)
				task.wait(0.05)
			end
		end)
	end
})

-- ===== ANTI AFK =====
SettingSection:NewToggle({
	Title = "Anti-AFK",
	Default = false,
	Callback = function(state)
		if not state then return end

		pcall(function()
			local Module = require(
				LocalPlayer
					.PlayerScripts
					.ClientMain
					.Replications
					.Workers
					.WalkDummy
			)

			setconstant(Module, 34, function()
				RunService.Heartbeat:Wait()
			end)
		end)
	end
})

-- ===== REJOIN =====

-- ===== HUMANOID FUNC =====
local function getHumanoid()
	local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
	return char:WaitForChild("Humanoid")
end

-- ===== SUPER JUMP =====
SettingSection:NewSlider({
	Title = "Super Jump",
	Min = 10,
	Max = 100,
	Default = 25,
	Callback = function(value)
		getHumanoid().JumpPower = value
	end
})

-- ===== WALK SPEED =====
